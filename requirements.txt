# -----------------------------------------------------------------------------
#  COMPUTER VISION  —  frame capture, display, and drawing
# -----------------------------------------------------------------------------

# OpenCV (Open Source Computer Vision Library)
#   Used for: webcam capture (VideoCapture), frame display (imshow),
#             drawing bounding boxes and labels (rectangle, putText, addWeighted).
#   Headless Linux servers should use opencv-python-headless instead.
opencv-python>=4.8.0


# -----------------------------------------------------------------------------
#  AI / VISION API  —  component recognition via Google Gemini
# -----------------------------------------------------------------------------

# Google Generative AI Python SDK
#   Used for: sending captured frames to Gemini Vision (gemini-1.5-flash or
#             gemini-1.5-pro) and receiving structured JSON detections.
#   Docs: https://ai.google.dev/gemini-api/docs/quickstart?lang=python
google-generativeai>=0.7.0


# -----------------------------------------------------------------------------
#  IMAGE PROCESSING  —  format conversion between OpenCV and Gemini
# -----------------------------------------------------------------------------

# Pillow (PIL fork)
#   Used for: converting OpenCV BGR numpy arrays to PIL RGB images before
#             sending to the Gemini multimodal API (which expects PIL.Image).
Pillow>=10.0.0


# -----------------------------------------------------------------------------
#  NUMERICAL COMPUTING  —  array manipulation and jitter calculations
# -----------------------------------------------------------------------------

# NumPy
#   Used for: frame representation as ndarray (returned by OpenCV),
#             random jitter offsets for bounding box de-overlap.
#   Note: OpenCV and Pillow both depend on NumPy; listed here explicitly
#         so the pinned minimum version is enforced across the project.
numpy>=1.24.0


# =============================================================================
